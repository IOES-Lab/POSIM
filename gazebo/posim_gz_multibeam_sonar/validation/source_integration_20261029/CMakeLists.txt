cmake_minimum_required(VERSION 3.22)
project(render_demand_probe LANGUAGES CXX)
find_package(gz-sim REQUIRED)
find_package(gz-plugin REQUIRED COMPONENTS register loader)
add_library(render-demand SHARED RenderDemand.cc)
target_compile_features(render-demand PRIVATE cxx_std_17)
target_link_libraries(render-demand PRIVATE gz-sim::gz-sim gz-plugin::register)
add_executable(loader-probe LoaderProbe.cc)
target_compile_features(loader-probe PRIVATE cxx_std_17)
target_link_libraries(loader-probe PRIVATE gz-plugin::loader)
add_library(dummy-plugin SHARED DummyPlugin.cc)
target_compile_features(dummy-plugin PRIVATE cxx_std_17)
target_link_libraries(dummy-plugin PRIVATE gz-plugin::register)
add_library(optional-lookup-plugin SHARED DummyPlugin.cc OptionalLookup.cc)
target_compile_features(optional-lookup-plugin PRIVATE cxx_std_17)
target_link_libraries(optional-lookup-plugin PRIVATE gz-plugin::register)
