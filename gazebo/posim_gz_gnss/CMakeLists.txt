cmake_minimum_required(VERSION 3.22)
project(posim_gz_gnss LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(gz-sim REQUIRED)
find_package(gz-plugin REQUIRED COMPONENTS register)
add_library(posim_gnss SHARED Gnss.cc)
target_compile_features(posim_gnss PRIVATE cxx_std_17)
target_link_libraries(posim_gnss PRIVATE gz-sim::gz-sim gz-plugin::register)
install(TARGETS posim_gnss LIBRARY DESTINATION lib)
install(DIRECTORY config/ DESTINATION share/posim/gnss)
install(FILES NOTICE README.md ../../LICENSE DESTINATION share/posim/gnss)
ament_environment_hooks(gnss.dsv.in)
ament_package()
